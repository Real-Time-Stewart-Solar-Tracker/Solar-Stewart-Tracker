28// Copy Y plane into packed grayscale buffer (width*height), respecting stride.
29static void copyPlaneY_StrideAware(std::vector<uint8_t>& outGray,
30 int fd,
31 std::size_t length,
32 std::size_t offset,
33 int width,
34 int height,
35 int strideBytes)
36{
37 if (fd < 0 || length == 0) {
38 throw std::runtime_error("Invalid buffer plane (fd/length)");
39 }
40 if (width <= 0 || height <= 0) {
41 throw std::runtime_error("Invalid dimensions");
42 }
43 if (strideBytes <= 0) {
44 throw std::runtime_error("Invalid stride");
45 }
46
47 void* mem = mmap(nullptr, length, PROT_READ, MAP_SHARED, fd, static_cast<off_t>(offset));
48 if (mem == MAP_FAILED) {
49 throw std::runtime_error("mmap failed for buffer plane");
50 }
51
52 const uint8_t* src = static_cast<const uint8_t*>(mem);
53 outGray.resize(static_cast<std::size_t>(width) * static_cast<std::size_t>(height));
54
55 for (int y = 0; y < height; ++y) {
56 const std::size_t srcRow = static_cast<std::size_t>(y) * static_cast<std::size_t>(strideBytes);
57 const std::size_t dstRow = static_cast<std::size_t>(y) * static_cast<std::size_t>(width);
58
59 if (srcRow + static_cast<std::size_t>(width) > length) {
60 munmap(mem, length);
61 throw std::runtime_error("Plane copy out of bounds (stride/length mismatch)");
62 }
63
64 std::memcpy(outGray.data() + dstRow, src + srcRow, static_cast<std::size_t>(width));
65 }
66
67 munmap(mem, length);
68}
