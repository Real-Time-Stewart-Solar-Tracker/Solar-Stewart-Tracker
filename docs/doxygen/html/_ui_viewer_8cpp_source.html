35// Draw text with a semi-transparent background box (stable + readable)
36static inline void drawTextBox(cv::Mat& img,
37 const std::string& text,
38 cv::Point org, // baseline start
39 int fontFace,
40 double fontScale,
41 const cv::Scalar& textColor,
42 int thickness,
43 const cv::Scalar& boxColor,
44 double boxAlpha = 0.55) {
45 int base = 0;
46 const cv::Size sz = cv::getTextSize(text, fontFace, fontScale, thickness, &base);
47
48 const int padX = 10;
49 const int padY = 7;
50
51 cv::Rect r(org.x - padX,
52 org.y - sz.height - padY,
53 sz.width + 2 * padX,
54 sz.height + base + 2 * padY);
55
56 // clamp inside image
57 r.x = std::max(0, r.x);
58 r.y = std::max(0, r.y);
59 r.width = std::min(r.width, img.cols - r.x);
60 r.height = std::min(r.height, img.rows - r.y);
61
62 cv::Mat roi = img(r);
63 cv::Mat overlay;
64 roi.copyTo(overlay);
65 cv::rectangle(overlay, cv::Rect(0, 0, r.width, r.height), boxColor, cv::FILLED);
66 cv::addWeighted(overlay, boxAlpha, roi, 1.0 - boxAlpha, 0.0, roi);
67
68 cv::putText(img, text, org, fontFace, fontScale, textColor, thickness, cv::LINE_AA);
69}

109

124void UiViewer::onCommand(const ActuatorCommand& cmd) {
125 std::lock_guard<std::mutex> lock(m_);
126 lastCmd_ = cmd;
127
128 // Advance ring buffer clock here (one sample per actuator update)
129 pushSample_(cmd.actuator_targets[0],
130 cmd.actuator_targets[1],
131 cmd.actuator_targets[2]);
132
133 // Align latest estimate to this same sample slot
134 if (!hist_cx_.empty()) {
135 const int N = static_cast<int>(hist_cx_.size());
136 const int idx = (hist_head_ - 1 + N) % N;
137 hist_cx_[idx] = lastEst_.cx;
138 hist_cy_[idx] = lastEst_.cy;
139 }
140
141 // Align latest latency to this same slot
142 if (!hist_lat_[0].empty()) {
143 const int N = static_cast<int>(hist_lat_[0].size());
144 const int idx = (hist_head_ - 1 + N) % N;
145 hist_lat_[0][idx] = lastLat_.cap_to_est_ms;
146 hist_lat_[1][idx] = lastLat_.est_to_ctrl_ms;
147 hist_lat_[2][idx] = lastLat_.ctrl_to_act_ms;
148 }
149}

172void UiViewer::handleMouse_(int event) {
173 int thr = threshold_.load();
174
175 if (event == cv::EVENT_LBUTTONDOWN) {
176 thr += cfg_.thr_step;
177 } else if (event == cv::EVENT_RBUTTONDOWN) {
178 thr -= cfg_.thr_step;
179 } else {
180 return;
181 }
182
183 thr = clampi(thr, 0, 255);
184 threshold_.store(thr);
185 log_.info("UI threshold -> " + std::to_string(thr));
186}

188// -------------------- ring buffer helpers --------------------
189void UiViewer::pushSample_(float v0, float v1, float v2) {
190 const int N = static_cast<int>(hist_u_[0].size());
191 if (N <= 0) return;
192
193 hist_u_[0][hist_head_] = v0;
194 hist_u_[1][hist_head_] = v1;
195 hist_u_[2][hist_head_] = v2;
196
197 hist_head_ = (hist_head_ + 1) % N;
198 hist_count_ = std::min(hist_count_ + 1, N);
199}

289void UiViewer::drawPlots_(cv::Mat& vis) {
290 const int H = std::max(140, cfg_.plot_height);
291 const int W = vis.cols;
292
293 cv::Rect plotRect(0, vis.rows - H, W, H);
294 cv::rectangle(vis, plotRect, cv::Scalar(255,255,255), cv::FILLED);
295
296 // layout
297 const int outerL = 62;
298 const int outerR = 14;
299 const int outerT = 10;
300 const int outerB = 10;
301 const int gap = 12;
302 const int titleH = 20;
303
304 const int usableW = plotRect.width - outerL - outerR;
305 const int usableH = plotRect.height - outerT - outerB;
306
307 const int h1 = (usableH - gap) * 2 / 3; // actuators
308 const int h2 = (usableH - gap) - h1; // latency
309
310 cv::Rect ax1(plotRect.x + outerL,
311 plotRect.y + outerT + titleH,
312 usableW,
313 h1 - titleH);
314
315 cv::Rect ax2(plotRect.x + outerL,
316 ax1.y + ax1.height + gap + titleH,
317 usableW,
318 h2 - titleH);
319
320 auto drawGridBox = [&](const cv::Rect& ax) {
321 cv::rectangle(vis, ax, cv::Scalar(0,0,0), 1);
322 const int gy = 4;
323 for (int i = 1; i <= gy; ++i) {
324 const int y = ax.y + (i * ax.height) / (gy + 1);
325 cv::line(vis, {ax.x, y}, {ax.x + ax.width, y}, cv::Scalar(230,230,230), 1);
326 }
327 const int gx = 6;
328 for (int i = 1; i <= gx; ++i) {
329 const int x = ax.x + (i * ax.width) / (gx + 1);
330 cv::line(vis, {x, ax.y}, {x, ax.y + ax.height}, cv::Scalar(242,242,242), 1);
331 }
332 };
333
334 if (hist_count_ < 2) {
335 drawTextBox(vis, "Waiting for data...",
336 {plotRect.x + 20, plotRect.y + 32},
337 cv::FONT_HERSHEY_SIMPLEX, 0.7,
338 {0,0,0}, 2, {255,255,255}, 0.0);
339 return;
340 }
341
342 auto idxAt = [&](int back) {
343 const int N = static_cast<int>(hist_u_[0].size());
344 int idx = hist_head_ - 1 - back;
345 while (idx < 0) idx += N;
346 return idx % N;
347 };
348
349 const int stride = std::max(1, cfg_.plot_stride_px);
350 const int maxSamples = std::min(hist_count_, std::min(ax1.width, ax2.width) / stride);
351
352 auto xOf = [&](const cv::Rect& ax, int back) -> int {
353 const int xi = ax.x + (ax.width - 1) - back * stride;
354 return std::clamp(xi, ax.x, ax.x + ax.width - 1);
355 };
356
357 auto yMap = [&](const cv::Rect& ax, float v, float mn, float mx) -> int {
358 const float t = (v - mn) / (mx - mn);
359 const float yf = static_cast<float>(ax.y) + (1.0f - t) * static_cast<float>(ax.height - 1);
360 const int yi = static_cast<int>(std::lround(yf));
361 return std::clamp(yi, ax.y, ax.y + ax.height - 1);
362 };
363
364 auto drawYTicks = [&](const cv::Rect& ax, float mn, float mx) {
365 const int ticks = 4;
366 for (int i = 0; i <= ticks; ++i) {
367 const float a = static_cast<float>(i) / static_cast<float>(ticks);
368 const float val = mx - a * (mx - mn);
369 const int y = ax.y + static_cast<int>(a * (ax.height - 1));
370
371 cv::line(vis, {ax.x - 5, y}, {ax.x, y}, cv::Scalar(0,0,0), 1);
372 cv::putText(vis, fmt(val, 3), {plotRect.x + 6, y + 4},
373 cv::FONT_HERSHEY_SIMPLEX, 0.45, cv::Scalar(0,0,0), 1, cv::LINE_AA);
374 }
375 };
376
377 // -------- subplot 1 (actuators) --------
378 float mnU = std::numeric_limits<float>::infinity();
379 float mxU = -std::numeric_limits<float>::infinity();
380 for (int i = 0; i < maxSamples; ++i) {
381 const int idx = idxAt(i);
382 for (int k = 0; k < 3; ++k) {
383 mnU = std::min(mnU, hist_u_[k][idx]);
384 mxU = std::max(mxU, hist_u_[k][idx]);
385 }
386 }
387 if (!std::isfinite(mnU) || !std::isfinite(mxU) || mnU == mxU) { mnU = 0.f; mxU = 1.f; }
388 const float rU = mxU - mnU;
389 mnU -= 0.08f * rU; mxU += 0.08f * rU;
390
391 drawGridBox(ax1);
392 drawYTicks(ax1, mnU, mxU);
393
394 cv::putText(vis, "Actuator Targets (u0, u1, u2)",
395 {ax1.x, ax1.y - 6},
396 cv::FONT_HERSHEY_SIMPLEX, 0.55, cv::Scalar(0,0,0), 1, cv::LINE_AA);
397
398 // legend (spaced, in title line)
399 cv::putText(vis, "u0", {ax1.x + 260, ax1.y - 6}, cv::FONT_HERSHEY_SIMPLEX, 0.55, cv::Scalar(0,255,255), 2, cv::LINE_AA);
400 cv::putText(vis, "u1", {ax1.x + 305, ax1.y - 6}, cv::FONT_HERSHEY_SIMPLEX, 0.55, cv::Scalar(255,255,0), 2, cv::LINE_AA);
401 cv::putText(vis, "u2", {ax1.x + 350, ax1.y - 6}, cv::FONT_HERSHEY_SIMPLEX, 0.55, cv::Scalar(255,0,255), 2, cv::LINE_AA);
402
403 const cv::Scalar colU[3] = { {0,255,255}, {255,255,0}, {255,0,255} };
404 for (int k = 0; k < 3; ++k) {
405 cv::Point prev; bool hasPrev = false;
406 for (int i = 0; i < maxSamples; ++i) {
407 const int idx = idxAt(i);
408 const int x = xOf(ax1, i);
409 const int y = yMap(ax1, hist_u_[k][idx], mnU, mxU);
410 const cv::Point p(x, y);
411 if (hasPrev) cv::line(vis, prev, p, colU[k], 2, cv::LINE_AA);
412 prev = p; hasPrev = true;
413 }
414 }
415
416 // -------- subplot 2 (latency) --------
417 float mnL = std::numeric_limits<float>::infinity();
418 float mxL = -std::numeric_limits<float>::infinity();
419 for (int i = 0; i < maxSamples; ++i) {
420 const int idx = idxAt(i);
421 for (int k = 0; k < 3; ++k) {
422 mnL = std::min(mnL, hist_lat_[k][idx]);
423 mxL = std::max(mxL, hist_lat_[k][idx]);
424 }
425 }
426 if (!std::isfinite(mnL) || !std::isfinite(mxL) || mnL == mxL) { mnL = 0.f; mxL = 50.f; }
427 const float rL = mxL - mnL;
428 mnL -= 0.08f * rL; mxL += 0.08f * rL;
429
430 drawGridBox(ax2);
431 drawYTicks(ax2, mnL, mxL);
432
433 cv::putText(vis, "Latency (ms): cap->est, est->ctrl, ctrl->act",
434 {ax2.x, ax2.y - 6},
435 cv::FONT_HERSHEY_SIMPLEX, 0.55, cv::Scalar(0,0,0), 1, cv::LINE_AA);
436
437 // legend (spaced, no overlap)
438 const int lx = ax2.x + 260;
439 const int ly = ax2.y - 6;
440 cv::putText(vis, "cap->est", {lx, ly}, cv::FONT_HERSHEY_SIMPLEX, 0.50, cv::Scalar(0,0,255), 2, cv::LINE_AA);
441 cv::putText(vis, "est->ctrl", {lx + 120, ly}, cv::FONT_HERSHEY_SIMPLEX, 0.50, cv::Scalar(255,0,0), 2, cv::LINE_AA);
442 cv::putText(vis, "ctrl->act", {lx + 260, ly}, cv::FONT_HERSHEY_SIMPLEX, 0.50, cv::Scalar(0,150,0), 2, cv::LINE_AA);
443
444 const cv::Scalar colL[3] = { {0,0,255}, {255,0,0}, {0,150,0} };
445 for (int k = 0; k < 3; ++k) {
446 cv::Point prev; bool hasPrev = false;
447 for (int i = 0; i < maxSamples; ++i) {
448 const int idx = idxAt(i);
449 const int x = xOf(ax2, i);
450 const int y = yMap(ax2, hist_lat_[k][idx], mnL, mxL);
451 const cv::Point p(x, y);
452 if (hasPrev) cv::line(vis, prev, p, colL[k], 1, cv::LINE_AA);
453 prev = p; hasPrev = true;
454 }

458// -------------------- tick --------------------
459bool UiViewer::tick() {
460 cv::Mat grayCopy;
461 SunEstimate estCopy;
462
463 {
464 std::lock_guard<std::mutex> lock(m_);
465 if (!gray_.empty()) gray_.copyTo(grayCopy);
466 estCopy = lastEst_;
467 }
468
469 if (grayCopy.empty()) {
470 cv::Mat blank(cfg_.height, cfg_.width, CV_8UC3, cv::Scalar(0,0,0));
471 cv::putText(blank, "Waiting for frames...",
472 {20, 40}, cv::FONT_HERSHEY_SIMPLEX, 0.8,
473 cv::Scalar(200,200,200), 2, cv::LINE_AA);
474 cv::imshow(cfg_.window_name, blank);
475 } else {
476 const int plotH = std::max(140, cfg_.plot_height);
477 cv::Mat vis(cfg_.height + plotH, cfg_.width, CV_8UC3, cv::Scalar(0,0,0));
478
479 cv::Mat top = vis(cv::Rect(0, 0, cfg_.width, cfg_.height));
480 cv::cvtColor(grayCopy, top, cv::COLOR_GRAY2BGR);
481
482 // centroid
483 if (estCopy.confidence > 0.0f) {
484 cv::circle(top,
485 cv::Point(static_cast<int>(std::lround(estCopy.cx)),
486 static_cast<int>(std::lround(estCopy.cy))),
487 8, cv::Scalar(0,255,255), 2, cv::LINE_AA);
488 }
489
490 drawOverlay_(top, estCopy.confidence, threshold_.load());
491 drawPlots_(vis);
492
493 cv::imshow(cfg_.window_name, vis);
494 }
495
496 const int key = cv::waitKey(1);
497 if (key == 27 || key == 'q' || key == 'Q') {
498 quit_ = true;
499 }
500 return !quit_;
501}
