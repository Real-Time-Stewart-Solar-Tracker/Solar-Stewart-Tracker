23// Per-frame timing cache for live latency computation (ms)
24struct Times {
25 std::chrono::steady_clock::time_point t_cap{};
26 std::chrono::steady_clock::time_point t_est{};
27 std::chrono::steady_clock::time_point t_ctrl{};
28 std::chrono::steady_clock::time_point t_act{};
29 bool has_cap{false};
30 bool has_est{false};
31 bool has_ctrl{false};
32 bool has_act{false};
33};
34
35// Keep it small and bounded: erase on completion, and also cap size.
36std::mutex g_lat_mtx;
37std::unordered_map<uint64_t, Times> g_times;
38
39static inline float msBetween(const std::chrono::steady_clock::time_point& a,
40 const std::chrono::steady_clock::time_point& b) {
41 return std::chrono::duration_cast<std::chrono::duration<float, std::milli>>(b - a).count();
42}
43
44static inline void capMapSize() {
45 // Very simple cap to prevent unbounded growth if something goes wrong.
46 constexpr std::size_t MAX_FRAMES = 4096;
47 if (g_times.size() <= MAX_FRAMES) return;
48
49 // Erase arbitrary old entries (not ordered). Good enough for safety.
50 auto it = g_times.begin();
51 const std::size_t toErase = g_times.size() - MAX_FRAMES;
52 for (std::size_t i = 0; i < toErase && it != g_times.end(); ++i) {
53 it = g_times.erase(it);
54 }
55}
56
57static inline void tryEmitLatency(uint64_t frame_id) {
58 SystemManager::LatencyObserver cb;
59 {
60 std::lock_guard<std::mutex> lk(g_obs_mtx);
61 cb = g_lat_obs;
62 }
63 if (!cb) return;
64
65 Times t{};
66 {
67 std::lock_guard<std::mutex> lk(g_lat_mtx);
68 auto it = g_times.find(frame_id);
69 if (it == g_times.end()) return;
70 t = it->second;
71
72 // Emit only when we have the full chain
73 if (!(t.has_cap && t.has_est && t.has_ctrl && t.has_act)) return;
74
75 // Done -> erase immediately
76 g_times.erase(it);
77 }
78
79 const float cap_to_est = msBetween(t.t_cap, t.t_est);
80 const float est_to_ctl = msBetween(t.t_est, t.t_ctrl);
81 const float ctl_to_act = msBetween(t.t_ctrl, t.t_act);
82
83 cb(frame_id, cap_to_est, est_to_ctl, ctl_to_act);
84}

104 driver_(log_, drvCfg),
105 latency_(log_),
106 min_confidence_(controllerCfg.min_confidence) {
107
108 // Camera -> SunTracker (+ frame observer)
109 if (camera_) {
110 camera_->registerFrameCallback([this](const FrameEvent& fe) {
111 latency_.onCapture(fe.frame_id, fe.t_capture);
112
113 // cache capture time for live latency
114 {
115 std::lock_guard<std::mutex> lk(g_lat_mtx);
116 auto& ts = g_times[fe.frame_id];
117 ts.t_cap = fe.t_capture;
118 ts.has_cap = true;
119 capMapSize();
120 }
121
122 // observer hook
123 ICamera::FrameCallback cb;
124 {
125 std::lock_guard<std::mutex> lk(g_obs_mtx);
126 cb = g_frame_obs;
127 }
128 if (cb) cb(fe);
129
130 tracker_.onFrame(fe);
131 });
132 }
133
134 // SunTracker -> Controller (+ estimate observer, + SEARCHING/TRACKING transitions)
135 tracker_.registerEstimateCallback([this](const SunEstimate& est) {
136 latency_.onEstimate(est.frame_id, est.t_estimate);
137
138 // cache estimate time for live latency
139 {
140 std::lock_guard<std::mutex> lk(g_lat_mtx);
141 auto& ts = g_times[est.frame_id];
142 ts.t_est = est.t_estimate;
143 ts.has_est = true;
144 capMapSize();
145 }
146
147 // observer hook
148 SunTracker::EstimateCallback cb;
149 {
150 std::lock_guard<std::mutex> lk(g_obs_mtx);
151 cb = g_est_obs;
152 }
153 if (cb) cb(est);
154
155 // confidence drives SEARCHING/TRACKING (unless MANUAL/FAULT)
156 if (running_ && state_ != TrackerState::MANUAL && state_ != TrackerState::FAULT) {
157 if (est.confidence >= min_confidence_) setState_(TrackerState::TRACKING);
158 else setState_(TrackerState::SEARCHING);
159 }
160
161 // In MANUAL mode, ignore AUTO controller updates
162 if (state_ != TrackerState::MANUAL) {
163 controller_.onEstimate(est);
164 }
165 });
166
167 // Controller -> Kinematics (+ setpoint observer)
168 controller_.registerSetpointCallback([this](const PlatformSetpoint& sp) {
169 latency_.onControl(sp.frame_id, sp.t_control);
170
171 // cache control time for live latency
172 {
173 std::lock_guard<std::mutex> lk(g_lat_mtx);
174 auto& ts = g_times[sp.frame_id];
175 ts.t_ctrl = sp.t_control;
176 ts.has_ctrl = true;
177 capMapSize();
178 }
179
180 // observer hook
181 Controller::SetpointCallback cb;
182 {
183 std::lock_guard<std::mutex> lk(g_obs_mtx);
184 cb = g_setpoint_obs;
185 }
186 if (cb) cb(sp);
187
188 kinematics_.onSetpoint(sp);
189 });
190
191 // Kinematics -> ActuatorManager (+ command observer)
192 kinematics_.registerCommandCallback([this](const ActuatorCommand& cmd) {
193 // observer hook
194 Kinematics3RRS::CommandCallback cb;
195 {
196 std::lock_guard<std::mutex> lk(g_obs_mtx);
197 cb = g_cmd_obs;
198 }
199 if (cb) cb(cmd);
200
201 actuatorMgr_.onCommand(cmd);
202 });
203
204 // ActuatorManager -> ServoDriver (+ latency complete)
205 actuatorMgr_.registerSafeCommandCallback([this](const ActuatorCommand& safeCmd) {
206 latency_.onActuate(safeCmd.frame_id, safeCmd.t_actuate);
207
208 // cache actuate time for live latency, then try emit
209 {
210 std::lock_guard<std::mutex> lk(g_lat_mtx);
211 auto& ts = g_times[safeCmd.frame_id];
212 ts.t_act = safeCmd.t_actuate;
213 ts.has_act = true;
214 capMapSize();
215 }
216
217 tryEmitLatency(safeCmd.frame_id);
218
219 driver_.apply(safeCmd);
220 });
221}

227bool SystemManager::start() {
228 if (running_) return true;
229
230 setState_(TrackerState::STARTUP);
231
232 if (!camera_) {
233 log_.error("SystemManager: camera is null");
234 setState_(TrackerState::FAULT);
235 return false;
236 }
237
238 if (!driver_.start()) {
239 log_.error("SystemManager: ServoDriver start failed");
240 setState_(TrackerState::FAULT);
241 return false;
242 }
243
244 // IMPORTANT: set running_ before camera starts producing callbacks
245 running_ = true;
246
247 if (!camera_->start()) {
248 log_.error("SystemManager: Camera start failed");
249 running_ = false;
250 driver_.stop();
251 setState_(TrackerState::FAULT);
252 return false;
253 }
254
255 setState_(TrackerState::NEUTRAL);
256 applyNeutralOnce_();
257 setState_(TrackerState::SEARCHING);
258
259 log_.info("SystemManager started");
260 return true;
261}

263void SystemManager::stop() {
264 if (!running_) return;
265
266 setState_(TrackerState::STOPPING);
267
268 applyNeutralOnce_();
269
270 if (camera_) camera_->stop();
271 driver_.stop();
272
273 running_ = false;
274
275 latency_.printSummary();
276
277 setState_(TrackerState::IDLE);
278 log_.info("SystemManager stopped");
279}
