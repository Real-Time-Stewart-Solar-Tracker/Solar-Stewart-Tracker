32void Kinematics3RRS::computeIK_(const PlatformSetpoint& sp) {
33 // Mapping:
34 // - sp.tilt_rad -> pitch
35 // - sp.pan_rad -> roll
36 const float pitch = sp.tilt_rad;
37 const float roll = sp.pan_rad;
38
39 // Rotation matrix R = Ry(pitch) * Rx(roll) (yaw fixed to 0)
40 const float cp = std::cos(pitch);
41 const float spc = std::sin(pitch);
42 const float cr = std::cos(roll);
43 const float sr = std::sin(roll);
44
45 float R[3][3];
46 R[0][0] = cp;
47 R[0][1] = spc * sr;
48 R[0][2] = spc * cr;
49
50 R[1][0] = 0.f;
51 R[1][1] = cr;
52 R[1][2] = -sr;
53
54 R[2][0] = -spc;
55 R[2][1] = cp * sr;
56 R[2][2] = cp * cr;
57
58 const float h = cfg_.home_height_m;
59 const float Rb = cfg_.base_radius_m;
60 const float Rp = cfg_.platform_radius_m;
61 const float L1 = cfg_.horn_length_m;
62 const float L2 = cfg_.rod_length_m;
63
64 ActuatorCommand cmd;
65 cmd.frame_id = sp.frame_id;
66 cmd.t_actuate = std::chrono::steady_clock::now();
67
68 // Basic parameter sanity guard (prevents NaNs)
69 if (L1 <= 0.0f || L2 <= 0.0f || Rb <= 0.0f || Rp <= 0.0f) {
70 log_.warn("Kinematics3RRS: invalid geometry config (non-positive length/radius)");
71 // Fail-safe: hold previous commanded angles (already in q_prev_ with neutral applied below)
72 for (int i = 0; i < 3; ++i) {
73 const std::size_t idx = static_cast<std::size_t>(i);
74 float chosen = q_prev_[idx];
75 chosen = static_cast<float>(cfg_.servo_dir[idx]) * chosen + cfg_.servo_neutral_rad[idx];

77 }

90 for (int i = 0; i < 3; ++i) {
91 const std::size_t idx = static_cast<std::size_t>(i);
92
93 // Base joint B_i
94 const float thb = deg2rad(cfg_.base_theta_deg[idx]);
95 const float Bx = Rb * std::cos(thb);
96 const float By = Rb * std::sin(thb);
97 const float Bz = 0.f;
98
99 // Platform joint P_i in platform local
100 const float thp = deg2rad(cfg_.plat_theta_deg[idx]);
101 const float Px_l = Rp * std::cos(thp);
102 const float Py_l = Rp * std::sin(thp);
103 const float Pz_l = 0.f;
104
105 // Rotate + translate
106 float Px = R[0][0] * Px_l + R[0][1] * Py_l + R[0][2] * Pz_l;
107 float Py = R[1][0] * Px_l + R[1][1] * Py_l + R[1][2] * Pz_l;
108 float Pz = R[2][0] * Px_l + R[2][1] * Py_l + R[2][2] * Pz_l;
109
110 Pz += h;
111
112 // Vector base->platform
113 const float dx = Px - Bx;
114 const float dy = Py - By;
115 const float dz = Pz - Bz;
116
117 // Projection along base leg plane direction
118 const float C = dx * std::cos(thb) + dy * std::sin(thb);
119
120 const float Rxz = std::sqrt(C * C + dz * dz);
121
122 // Guard against degenerate geometry / division by zero
123 if (Rxz <= EPS) {
124 log_.warn("Kinematics3RRS: degenerate Rxz; holding previous angle");
125 float chosen = q_prev_[idx];
126 chosen = static_cast<float>(cfg_.servo_dir[idx]) * chosen + cfg_.servo_neutral_rad[idx];
127 cmd.actuator_targets[idx] = chosen;
128 continue;
129 }
130
131 float D = (L1 * L1 + Rxz * Rxz - L2 * L2) / (2.f * L1 * Rxz);
132 D = std::clamp(D, -1.f, 1.f);
133
134 const float alpha = std::atan2(dz, C);
135 const float beta = std::acos(D);
136
137 const float q1 = alpha - beta;
138 const float q2 = alpha + beta;
139
140 // Choose branch closest to previous
141 const float prev = q_prev_[idx];
142 float chosen = (std::abs(q1 - prev) < std::abs(q2 - prev)) ? q1 : q2;
143
144 q_prev_[idx] = chosen;
145
146 // Apply direction and neutral offsets
147 chosen = static_cast<float>(cfg_.servo_dir[idx]) * chosen
148 + cfg_.servo_neutral_rad[idx];
149
150 cmd.actuator_targets[idx] = chosen;
151 }
152
153 CommandCallback cb;
154 {

156 cb = cmdCb_;
157 }
158 if (cb) cb(cmd);
159}
